cmake_minimum_required(VERSION 3.20)
project(PowerRangersRuntimeTests C)
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/framework_patches.cmake")
power_rangers_patch_framework("${CMAKE_CURRENT_SOURCE_DIR}/../snesrecomp")
enable_testing()
set(CMAKE_C_STANDARD 11)
set(RUNNER "${CMAKE_CURRENT_SOURCE_DIR}/../snesrecomp/runner/src")
foreach(test ppu_display_memory ppu_raster_journal)
  if(test STREQUAL "ppu_display_memory")
    set(test_source "${CMAKE_CURRENT_SOURCE_DIR}/ppu_display_memory_test.c")
  else()
    set(test_source "${RUNNER}/../../tests/ppu/ppu_raster_journal_test.c")
  endif()
  add_executable(${test} "${test_source}" "${RUNNER}/snes/ppu.c"
    "${RUNNER}/snes/ppu_legacy.c")
  target_include_directories(${test} PRIVATE "${RUNNER}" "${RUNNER}/snes")
  target_compile_definitions(${test} PRIVATE SNESRECOMP_REVERSE_DEBUG=0)
  if(NOT MSVC)
    target_link_libraries(${test} PRIVATE m)
  endif()
  add_test(NAME ${test} COMMAND ${test})
endforeach()
